# Runtime dependencies for the Django backend.
#
# Every entry here is imported by code in this repository. Nothing is listed
# speculatively. If you add an import, add its package here in the same commit;
# if you remove the last import of a package, remove it here.
#
# Install (from the repository root, with your virtualenv active):
#     pip install -r backend/requirements.txt
#     pip install -e ./ml
#
# The second command installs the local `labelextract` package in editable
# mode. It is not listed below because a relative path in a requirements file
# resolves against your current working directory, which silently breaks when
# pip is run from anywhere else. An explicit second command fails loudly
# instead of installing the wrong thing.
#
# Version strategy: pinned to a compatible minor range (>=x.y,<x.y+1). Patch
# releases pick up security fixes automatically; a minor bump is a deliberate,
# reviewed change. Python 3.11+ is required.

# Web framework. LTS release.
Django>=5.2,<5.3

# REST layer: serializers, viewsets, versioning, throttling, exception handling.
djangorestframework>=3.16,<3.17

# CORS headers for the Vite dev server, which runs on a different origin.
# Configured in config/settings.py (CORS_ALLOWED_ORIGINS).
django-cors-headers>=4.9,<4.10

# PostgreSQL driver. psycopg 3 with prebuilt binaries, so no local C toolchain
# or libpq headers are needed - which matters for a six-person team on mixed
# operating systems.
psycopg[binary]>=3.2,<3.3

# Reads and type-casts environment variables in config/settings.py.
django-environ>=0.12,<0.13

# Image decoding for upload validation and metadata (apps/images/validators.py).
# Required, not speculative: validation works by actually decoding the file
# rather than trusting its extension, and that needs a real image library.
# This is NOT a machine-learning dependency - no OCR or ML framework is
# installed by this file.
Pillow>=11.3,<12

# --- Production serving -----------------------------------------------------
#
# Both are runtime dependencies of a deployment, not of development, but they
# belong here rather than in requirements-dev.txt: requirements-dev.txt exists
# so a deployment does not install a test runner, and these two are the exact
# opposite - a deployment that skipped them would have no server to run and no
# way to serve the admin's CSS.

# WSGI server. `manage.py runserver` is a development server and says so; it is
# single-threaded, does not restart a crashed worker, and is not written to
# face the internet. Gunicorn is configured in backend/gunicorn.conf.py.
gunicorn>=23.0,<24

# Serves STATIC_ROOT from inside the application process, so a deployment needs
# no second web server or CDN for the handful of files the Django admin uses.
# Wired in config/settings.py (middleware + STORAGES).
#
# It serves static files only. Uploaded product images are NOT served through
# it - see docs/deployment.md for what that means for media storage.
whitenoise>=6.9,<7

# --- OCR is deliberately NOT listed here -----------------------------------
#
# The Tesseract pipeline's Python dependency (pytesseract) lives in the `ocr`
# extra of ml/pyproject.toml, and the Tesseract binary is installed by your
# operating system's package manager. Neither belongs in this file:
#
#   * The backend imports `labelextract`, never an engine. Which engine runs is
#     resolved at runtime from DEFAULT_EXTRACTION_ENGINE_NAME, so a deployment
#     that uses the placeholder must not be made to install an OCR stack.
#   * pytesseract without the binary is useless, and pip cannot install the
#     binary. Listing it here would imply an install that pip cannot complete.
#
# To run OCR:
#     pip install -e "./ml[ocr]"
# plus the binary for your platform - see ml/README.md.
